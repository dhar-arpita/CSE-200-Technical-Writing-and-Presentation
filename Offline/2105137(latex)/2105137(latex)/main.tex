\section*{\textbf {\MakeUppercase{abstract}}}
 \noindent
We obtain robust and computationally efficient estimators for learning several linear models that achieve statistically optimal convergence rates under minimal distributional assumptions. Concretely, we assume our data is drawn from a $k$-hypercontractive distribution and an $\epsilon$-fraction is adversarially corrupted. We then describe an estimator that converges to the optimal least-squares minimizer for the true distribution at a rate proportional to $\epsilon^{2-2/k}$, when the noise is independent of the covariates. We note that no such estimator was known prior to our work, even with access to unbounded computation. The rate we achieve is information-theoretically optimal, and thus we resolve the main open question in Klivans, Kothari, and Meka [COLT'18].

Our key insight is to identify an analytic condition that serves as a polynomial relaxation of independence of random variables. In particular, we show that when the moments of the noise and covariates are negatively correlated, we obtain the same rate as independent noise. Further, when the condition is not satisfied, we obtain a rate proportional to $\epsilon^{2-4/k}$, and again match the information-theoretic lower bound. Our central technical contribution is to algorithmically exploit independence of random variables in the "sum-of-squares" framework by formulating it as the aforementioned polynomial inequality.

\section{\textbf{\MakeUppercase{introduction}}}
\noindent
This paper addresses the problem of robust linear regression, which involves creating regression models that remain accurate even when some data is corrupted by adversarial outliers. Traditional regression methods fail under these conditions, making robust estimation a challenging task. The authors focus on a hypercontractive distribution framework, which includes distributions like Gaussian, allowing them to design robust estimators. Their estimators achieve optimal convergence rates (best possible accuracy) even with corrupted data, something previous methods couldn’t do. Additionally, the estimators are computationally efficient, working in polynomial time using the sum-of-squares convex hierarchy. They also show that relying on just bounded covariance assumptions is insufficient for robust regression in adversarial settings. Their approach balances robustness, accuracy, and computational complexity, making it suitable for high-dimensional data. Prior methods, while statistically sound, were often computationally intractable in these scenarios. The paper’s key contribution is a unified framework that handles various noise models and outperforms previous estimators. Ultimately, the work provides a significant advancement in robust statistics by achieving both theoretical and practical efficiency.

\subsubsection{Model 1.1: Robust Regression Model}

Let $\mathcal{D}$ be a distribution over $\mathbb{R}^d \times \mathbb{R}$ such that the marginal distribution over $\mathbb{R}^d$ is centered and has covariance $\Sigma^\ast$, and let
\[
\Theta^\ast = \arg\min_{\Theta} \mathbb{E}_{(x,y) \sim \mathcal{D}} \left[ (y - \langle \Theta, x \rangle)^2 \right]
\]
be the optimal hyperplane for $\mathcal{D}$. Let $\{(x^\ast_1, y^\ast_1), (x^\ast_2, y^\ast_2), \dots, (x^\ast_n, y^\ast_n)\}$ be $n$ i.i.d. random variables drawn from $\mathcal{D}$. Given $\epsilon > 0$, the robust regression model $\mathcal{R}_{\mathcal{D}} (\epsilon, \Sigma^\ast, \Theta^\ast)$ outputs a set of $n$ samples $\{(x_i, y_i)\}_{i \in [n]}$ such that for at least $(1 - \epsilon)n$ points, $x_i = x^\ast_i$ and $y_i = y^\ast_i$. The remaining $\epsilon n$ points are arbitrary and potentially adversarial with respect to the input and estimator.

\subsubsection{Theorem 1.2: Bounded Covariance does not suffice (Theorem 6.3 informal)}

For all $\epsilon > 0$, there exist two distributions $\mathcal{D}_1$, $\mathcal{D}_2$ over $\mathbb{R}^d \times \mathbb{R}$ such that
\[
d_{TV} (\mathcal{D}_1, \mathcal{D}_2) \leq \epsilon
\]
and the marginal distribution over the covariates has bounded covariance, denoted by $\Sigma_2 = \Theta(1)$, yet
\[
\Sigma^{1/2} (\Theta_1 - \Theta_2)^2 = \Omega(1),
\]
where $\Theta_1$ and $\Theta_2$ are the optimal hyperplanes for $\mathcal{D}_1$ and $\mathcal{D}_2$, respectively.




\textbf{Definition 1.3} ($\mathcal{}{C,k}$-Hypercontractivity) A distribution \( D \)  over \( \{ -1, 1 \}^n \) is ($\mathcal{}{C,k}$)-hypercontractive for an even integer \( k \geq 2 \), [...]
\begin{equation}
r \in [k/2], \text{ for all } x \in \mathbb{R}^d,\hfill\break
\mathbb{E}_{x\sim D} \Biggr[ \Biggl \langle   v,x - \mathbb{E}[x])^2 \Biggr \rangle \Biggr]\leqslant r^2

\end{equation}



\textbf{Theorem 1.7 (Robust Regression with Negatively Correlated Noise, Theorem 5.1 informal).} 
Given $\epsilon > 0$, $k \geq 4$, and $n \geq (d \log(d))^{O(k)}$ samples from $\mathcal{R}_\mathcal{D}(\epsilon, \Sigma^\ast, \Theta^\ast)$, such that $\mathcal{D}$ is $(c, k)$-certifiably hypercontractive and has negatively correlated moments, there exists an algorithm that runs in $n^{O(k)}$ time and outputs an estimator $\tilde{\Theta}$ such that with high probability,
\[
(\Sigma^\ast)^{1/2} \left\|\Theta^\ast - \tilde{\Theta} \right\|_2 \leq O \left( \epsilon^{1 - 1/k} \right) \text{err}_{\mathcal{D}}(\Theta^\ast)^{1/2}.
\]

and , \\

\[
\text{err}_{\mathcal{D}}(\tilde{\Theta}) \leq \left( 1 + O\left( \epsilon^{2 - 2/k} \right) \right) \text{err}_{\mathcal{D}}(\Theta^\ast).
\]

\textbf{Theorem 1.9 (Lower Bound for Independent Noise, Theorem 6.1 informal).} For any $\epsilon > 0$, there exist two distributions $\mathcal{D}_1$, $\mathcal{D}_2$ over $\mathbb{R}^2 \times \mathbb{R}$ such that the marginal distribution over $\mathbb{R}^2$ has covariance $\Sigma$ and is $(c, k)$-hypercontractive for both distributions, and yet
\[
(\Sigma)^{1/2} \left\| \Theta_1 - \Theta_2 \right\|_2^2 = \Omega \left( \epsilon^{1 - 1/k} \sigma \right),
\]
where $\Theta_1$ and $\Theta_2$ are the optimal hyperplanes for $\mathcal{D}_1$ and $\mathcal{D}_2$, respectively, $\sigma = \max \left( \text{err}_{\mathcal{D}_1}(\Theta_1), \text{err}_{\mathcal{D}_2}(\Theta_2) \right)$, and the noise is uniform over $[-\sigma, \sigma]$. Further,
\[
\left| \text{err}_{\mathcal{D}_1}(\Theta_2) - \text{err}_{\mathcal{D}_1}(\Theta_1) \right| = \Omega \left( \epsilon^{2 - 2/k} \sigma^2 \right).
\]
.

\textbf{Corollary 1.10 (Robust Regression with Dependent Noise, Corollary 4.2 informal).} Given $\epsilon > 0$, $k \geq 4$, and $n \geq d^{O(k)}$ samples from $\mathcal{R}_\mathcal{D}(\epsilon, \Sigma^\ast, \Theta^\ast)$, such that $\mathcal{D}$ is $(c, k)$-certifiably hypercontractive, there exists an algorithm that runs in $n^{O(k)}$ time and outputs an estimator $\tilde{\Theta}$ such that with probability $9/10$,
\[
(\Sigma^\ast)^{1/2} \left\|\Theta^\ast - \tilde{\Theta} \right\|_2 \leq O \left( \epsilon^{1 - 2/k} \right) \text{err}_{\mathcal{D}}(\Theta^\ast)^{1/2},
\]
and
\[
\text{err}_{\mathcal{D}}(\tilde{\Theta}) \leq \left( 1 + O\left( \epsilon^{2 - 4/k} \right) \right) \text{err}_{\mathcal{D}}(\Theta^\ast).
\]

Further, we show that the dependence on $\epsilon$ is again information-theoretically optimal:

\textbf{Theorem 1.11 (Lower Bound for Dependent Noise, Theorem 6.2 informal).} For any $\epsilon > 0$, there exist two distributions $\mathcal{D}_1$, $\mathcal{D}_2$ over $\mathbb{R}^2 \times \mathbb{R}$ such that the marginal distribution over $\mathbb{R}^2$ has covariance $\Sigma$ and is $(c, k)$-hypercontractive for both distributions, and yet
\[
(\Sigma)^{1/2} \left\| \Theta_1 - \Theta_2 \right\|_2^2 = \Omega \left( \epsilon^{1 - 2/k} \sigma \right),
\]
where $\Theta_1$ and $\Theta_2$ are the optimal hyperplanes for $\mathcal{D}_1$ and $\mathcal{D}_2$, respectively, and $\sigma = \max \left( \text{err}_{\mathcal{D}_1}(\Theta_1), \text{err}_{\mathcal{D}_2}(\Theta_2) \right)$. Further,
\[
\left| \text{err}_{\mathcal{D}_1}(\Theta_2) - \text{err}_{\mathcal{D}_1}(\Theta_1) \right| = \Omega \left( \epsilon^{2 - 4/k} \sigma^2 \right).
\]

\textbf{Corollary 1.12 (Robust Regression with Gaussian Covariates).} Given $\epsilon > 0$ and $n \geq (d \log(d))^{O(\log(1/\epsilon))}$ samples from $\mathcal{R}_{\mathcal{N}}(\epsilon, \Sigma^\ast, \Theta^\ast)$, such that the marginal distribution over the $x$'s is $\mathcal{N}(0, \Sigma^\ast)$, there exists an algorithm that runs in $n^{O(\log(1/\epsilon))}$ time and outputs an estimator $\tilde{\Theta}$ such that with high probability,
\[
(\Sigma^\ast)^{1/2} \left\| \Theta^\ast - \tilde{\Theta} \right\|_2 \leq O \left( \epsilon \log(1/\epsilon) \right) \text{err}_{\mathcal{N}}(\Theta^\ast)^{1/2},
\]
and
\[
\text{err}_{\mathcal{N}}(\tilde{\Theta}) \leq \left( 1 + O \left( \left( \epsilon \log(1/\epsilon) \right)^2 \right) \right) \text{err}_{\mathcal{N}}(\Theta^\ast).
\]
We note that our estimators obtain the rate matching recent work for Gaussians, albeit in quasi-polynomial time. In comparison, Diakonikolas, Kong, and Stewart [21] obtain the same rate in polynomial time, when the noise is independent of the covariates. We note that obtaining the optimal rate for Gaussian covariates (shaving the additional $\log(1/\epsilon)$ factor) remains an outstanding open question.

\section{Robust Certifiability and Information Theoretic Estimators}

In this section, we provide an estimator that obtains the information theoretically optimal rate for robust regression. We note that we consider the setting where both the covariates and the noise are hypercontractive and are independent of each other. This setting displays all the key ideas of our estimator. Further, our estimator extends to the remaining settings, such as bounded dependent noise, by simple modifications to the subsequent analysis.

\textbf{Theorem 4.1 (Robust Certifiability with Optimal Rate).} Given \(\epsilon > 0\), let \(D\) and \(D'\) be distributions over \(\mathbb{R}^d \times \mathbb{R}\) such that the respective marginal distributions over \(\mathbb{R}^d\), denoted by \(D_X\) and \(D'_X\), are \((c k, k)\)-hypercontractive and \(\|D - D'\|_{\text{TV}} \leq \epsilon\). Let \(R_D (\epsilon, \Sigma_D, \Theta_D)\) and \(R_{D'} (\epsilon, \Sigma_{D'}, \Theta_{D'})\) be the corresponding instances of robust regression such that \(D\) and \(D'\) have negatively correlated moments. Further, for \((x, y) \sim D, D'\), let the marginal distribution over \(y - x\), \(\mathbb{E}[x x^\top]^{-1} \mathbb{E}[x y]\), be \((\eta k, k)\)-hypercontractive. Then,
\[
\|\Sigma_D^{1/2} (\Theta_D - \Theta_{D'})\|_2^2 \leq O \left( \sqrt{c k \eta k \epsilon^{1 - 1/k}} \left( \text{err}_D (\Theta_D)^{1/2} + \text{err}_{D'} (\Theta_{D'})^{1/2} \right) \right).
\]
Further,
\[
\text{err}_D (\Theta_{D'}) \leq \left( 1 + O \left( c k \eta k \epsilon^{2 - 2/k} \right) \right) \text{err}_D (\Theta_D) + O \left( c k \eta k \epsilon^{2 - 2/k} \right) \text{err}_{D'} (\Theta_{D'}).
\]

\textbf{Proof.} Consider a maximal coupling of \(D\) and \(D'\) over \((x, y) \times (x', y')\), denoted by \(G\), such that the marginal of \(G\) on \((x, y)\) is \(D\), the marginal on \((x', y')\) is \(D'\), and \(P_G \left[ I \{ (x, y) = (x', y') \} \right] = 1 - \epsilon\). Then, for all \(v\),
\[
\langle v, \Sigma_D (\Theta_D - \Theta_{D'}) \rangle = \mathbb{E}_G \left[ \langle v, x x^\top (\Theta_D - \Theta_{D'}) + x y - x' y' \rangle \right]
\]
\[
= \mathbb{E}_G \left[ \langle v, x (\langle x, \Theta_D \rangle - y) \rangle \right]
+ \mathbb{E}_G \left[ \langle v, x (y - \langle x, \Theta_{D'} \rangle) \rangle \right].
\]
Since \(\Theta_D\) is the minimizer for the least squares loss, we have the following gradient condition: for all \(v \in \mathbb{R}^d\),
\[
\mathbb{E}_{(x, y) \sim D} \left[ \langle v, (\langle x, \Theta_D \rangle - y) x \rangle \right] = 0.
\]

  \textbf{Algorithm: Optimal Robust Regression in Polynomial Time}\par
  \hspace{2em}\textbf{Input:} $n$ samples $\mathcal{X}_{\epsilon}$ from the robust regression model \\
     $\mathcal{R}_{\mathcal{D}}(\epsilon, \Theta^*, \Sigma^*)$.

  \hspace{4em}\textbf{Operation:}\vspace{0.5em}\hspace{5em}
  \begin{enumerate}
   \item Find a degree-$\mathcal{O}(k)$ pseudo-distribution $\tilde{\zeta}$ satisfying \\
   \quad  $\mathcal{A}_{\epsilon, \lambda_k}$ and minimizing
    \[
    \min_{w, x', y', \Theta} \mathbb{E}_{\tilde{\zeta}}\left[ \left( \frac{1}{n} \sum_{i \in [n]} w_i \left( y'_i - \langle \Theta, x'_i \rangle \right)^2 \right)^{k} \right].
    \]
    \item Round the pseudo-distribution to obtain an estimator $\mathbb{E}_{\tilde{\zeta}}[\Theta]$.
  \end{enumerate}

  \textbf{Output:} A vector $\mathbb{E}_{\tilde{\zeta}}[\Theta]$ such that the recovery guarantee in Theorem \cite{xu2013outlier} is satisfied.
\end{mdframed}
